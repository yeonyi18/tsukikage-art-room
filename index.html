const initialGoals = [ { id: "anatomy", name: "전신 인체 드로잉", target: 4, checked: [] }, { id: "forms", name: "인체 도형화 연습", target: 4, checked: [] }, { id: "gesture", name: "포즈 크로키", target: 4, checked: [] } ]; const storageKey = "tsukikage-art-goals-v1"; function loadGoals() { try { const saved = JSON.parse(localStorage.getItem(storageKey)); if (Array.isArray(saved) && saved.length) return saved; } catch (error) {} return initialGoals.map(goal => ({ ...goal })); } let goals = loadGoals(); function saveGoals() { try { localStorage.setItem(storageKey, JSON.stringify(goals)); } catch (error) {} } function render() { const total = goals.reduce((sum, goal) => sum + goal.target, 0); const completed = goals.reduce( (sum, goal) => sum + goal.checked.length, 0 ); const percent = total ? Math.round(completed / total * 100) : 0; document.getElementById("monthTitle").textContent = "이달의 수련"; document.getElementById("percent").textContent = `${percent}%`; document.getElementById("progressBar").style.width = `${percent}%`; document.getElementById("progressText").textContent = `완료 ${completed} / 목표 ${total}회`; const list = document.getElementById("goalList"); list.replaceChildren(); goals.forEach(goal => { const row = document.createElement("div"); row.className = "goal" + (goal.checked.length >= goal.target ? " done" : ""); const checkbox = document.createElement("input"); checkbox.type = "checkbox"; checkbox.disabled = true; checkbox.checked = goal.checked.length >= goal.target; checkbox.setAttribute("aria-label", goal.name + " 목표 완료"); const content = document.createElement("div"); content.className = "goal-content"; const name = document.createElement("input"); name.className = "goal-name"; name.value = goal.name; name.setAttribute("aria-label", "목표 이름 수정"); const count = document.createElement("span"); count.className = "goal-count"; count.textContent = `${goal.checked.length} / ${goal.target}회 완료 · 이름을 눌러 수정`; const controls = document.createElement("div"); controls.className = "goal-controls"; const minus = document.createElement("button"); minus.type = "button"; minus.textContent = "−"; minus.className = "count-button"; const plus = document.createElement("button"); plus.type = "button"; plus.textContent = "+"; plus.className = "count-button"; const counter = document.createElement("span"); counter.className = "count-number"; counter.textContent = `${goal.checked.length} / ${goal.target}`; minus.disabled = goal.checked.length === 0; plus.disabled = goal.checked.length >= goal.target; minus.addEventListener("click", () => { if (goal.checked.length > 0) { goal.checked.pop(); saveGoals(); render(); } }); plus.addEventListener("click", () => { if (goal.checked.length < goal.target) { goal.checked.push(goal.checked.length); saveGoals(); render(); } }); controls.append(minus, counter, plus); name.addEventListener("change", () => { goal.name = name.value.trim() || "새 수련 목표"; saveGoals(); render(); }); content.append(name, count, controls); row.append(checkbox, content); list.append(row); }); } document.getElementById("resetButton").addEventListener("click", () => { if (!confirm("이번 달 수련 체크를 모두 초기화할까?")) return; goals.forEach(goal => { goal.checked = []; }); saveGoals(); render(); }); render();
